# 后端微服务开发实施文档

## 1. 项目概述

本文档基于知识库管理系统的技术架构设计，详细说明如何使用Spring Boot + Kotlin构建微服务项目，采用DDD方法论和六边形架构，实现CQRS和Saga事务模式。

### 1.1 技术栈

- **框架**: Spring Boot 3.2+ + Kotlin 1.9+
- **数据库**: PostgreSQL 15 (关系数据 + 向量数据)
- **缓存**: Redis 7
- **消息队列**: Apache Kafka 3.5
- **CQRS实现**: 自定义实现CommandBus、QueryBus、EventBus
- **测试框架**: Kotest (BDD集成测试) + JUnit5 (单元测试)
- **容器化**: Docker + Kubernetes
- **监控**: Prometheus + Grafana + OpenTelemetry

### 1.2 微服务架构

系统按功能领域拆分为以下微服务，每个服务分为领域模块(内六边形)和应用模块(外六边形)：

1. **用户服务** (lifee-user + lifee-user-app)
2. **知识库服务** (lifee-knowledge + lifee-knowledge-app)
3. **对话服务** (lifee-chat + lifee-chat-app)
4. **推荐服务** (lifee-recommendation + lifee-recommendation-app)
5. **配置服务** (lifee-config + lifee-config-app)
6. **网关服务** (lifee-gateway)
7. **公共模块** (lifee-common)

```kotlin
// lifee-common/src/main/kotlin/com/lifee/common/cqrs/commands/CommandBus.kt
package com.lifee.common.cqrs.commands

interface CommandBus {
    fun <T : Command> send(command: T)
    fun <T : Command> sendAndWait(command: T)
}

// lifee-common/src/main/kotlin/com/lifee/common/cqrs/commands/DefaultCommandBus.kt
package com.lifee.common.cqrs.commands

import org.springframework.stereotype.Component
import java.util.concurrent.ConcurrentHashMap
import kotlin.reflect.KClass

@Component
class DefaultCommandBus : CommandBus {
    
    private val handlers = ConcurrentHashMap<KClass<out Command>, CommandHandler<*>>()
    
    fun <T : Command> registerHandler(commandType: KClass<T>, handler: CommandHandler<T>) {
        handlers[commandType] = handler
    }
    
    @Suppress("UNCHECKED_CAST")
    override fun <T : Command> send(command: T) {
        val handler = handlers[command::class] as? CommandHandler<T>
            ?: throw IllegalArgumentException("No handler found for command: ${command::class.simpleName}")
        
        handler.handle(command)
    }
    
    override fun <T : Command> sendAndWait(command: T) {
        send(command) // 同步执行
    }
}
```

#### 事件总线实现
```kotlin
// lifee-common/src/main/kotlin/com/lifee/common/cqrs/events/EventBus.kt
package com.lifee.common.cqrs.events

import com.lifee.common.domain.DomainEvent

interface EventBus {
    fun publish(event: DomainEvent)
    fun publishAll(events: List<DomainEvent>)
}

// lifee-common/src/main/kotlin/com/lifee/common/cqrs/events/DefaultEventBus.kt
package com.lifee.common.cqrs.events

import com.lifee.common.domain.DomainEvent
import org.springframework.kafka.core.KafkaTemplate
import org.springframework.stereotype.Component
import java.util.concurrent.ConcurrentHashMap
import kotlin.reflect.KClass

@Component
class DefaultEventBus(
    private val kafkaTemplate: KafkaTemplate<String, Any>
) : EventBus {
    
    private val localHandlers = ConcurrentHashMap<KClass<out DomainEvent>, MutableList<EventHandler<*>>>()
    
    fun <T : DomainEvent> registerHandler(eventType: KClass<T>, handler: EventHandler<T>) {
        localHandlers.computeIfAbsent(eventType) { mutableListOf() }.add(handler)
    }
    
    override fun publish(event: DomainEvent) {
        // 本地事件处理
        handleLocalEvent(event)
        
        // 发布到Kafka
        publishToKafka(event)
    }
    
    override fun publishAll(events: List<DomainEvent>) {
        events.forEach { publish(it) }
    }
    
    @Suppress("UNCHECKED_CAST")
    private fun handleLocalEvent(event: DomainEvent) {
        val handlers = localHandlers[event::class] as? List<EventHandler<DomainEvent>>
        handlers?.forEach { handler ->
            try {
                handler.handle(event)
            } catch (e: Exception) {
                // 记录错误日志，但不影响其他处理器
                println("Error handling event ${event::class.simpleName}: ${e.message}")
            }
        }
    }
    
    private fun publishToKafka(event: DomainEvent) {
        val topic = "${event::class.simpleName?.lowercase()}-events"
        kafkaTemplate.send(topic, event.aggregateId.toString(), event)
    }
}
```

```kotlin
// lifee-user/src/main/kotlin/com/lifee/user/domain/aggregates/User.kt
package com.lifee.user.domain.aggregates

import com.lifee.common.domain.AggregateRoot
import com.lifee.user.domain.valueobjects.*
import com.lifee.user.domain.events.*
import com.lifee.user.domain.exceptions.*
import com.lifee.user.domain.services.PasswordService
import java.time.Instant

class User private constructor(
    private val userId: UserId,
    private var email: Email,
    private var username: Username,
    private var passwordHash: PasswordHash,
    private var isActive: Boolean = false,
    private var createdAt: Instant,
    private var updatedAt: Instant
) : AggregateRoot<UserId>(userId) {
    
    companion object {
        fun create(
            userId: UserId,
            email: Email,
            username: Username,
            passwordHash: PasswordHash
        ): User {
            // 业务规则验证
            validateUserCreation(email, username)
            
            val now = Instant.now()
            val user = User(
                userId = userId,
                email = email,
                username = username,
                passwordHash = passwordHash,
                isActive = false,
                createdAt = now,
                updatedAt = now
            )
            
            // 发布领域事件
            user.addDomainEvent(
                UserRegisteredEvent(
                    userId = userId,
                    email = email,
                    username = username,
                    passwordHash = passwordHash,
                    createdAt = now
                )
            )
            
            return user
        }
        
        private fun validateUserCreation(email: Email, username: Username) {
            if (!email.isValid()) {
                throw InvalidEmailException(email.value)
            }
            if (!username.isValid()) {
                throw InvalidUsernameException(username.value)
            }
        }
    }
    
    fun activate() {
        if (isActive) {
            throw UserAlreadyActivatedException(userId)
        }
        
        val now = Instant.now()
        this.isActive = true
        this.updatedAt = now
        
        addDomainEvent(
            UserActivatedEvent(
                userId = userId,
                activatedAt = now
            )
        )
    }
    
    fun updateProfile(newEmail: Email?, newUsername: Username?) {
        val now = Instant.now()
        var hasChanges = false
        
        newEmail?.let {
            if (it != this.email) {
                this.email = it
                hasChanges = true
            }
        }
        
        newUsername?.let {
            if (it != this.username) {
                this.username = it
                hasChanges = true
            }
        }
        
        if (hasChanges) {
            this.updatedAt = now
            addDomainEvent(
                UserProfileUpdatedEvent(
                    userId = userId,
                    email = email,
                    username = username,
                    updatedAt = now
                )
            )
        }
    }
    
    fun changePassword(oldPassword: String, newPassword: String, passwordService: PasswordService) {
        if (!passwordService.verify(oldPassword, passwordHash)) {
            throw InvalidPasswordException()
        }
        
        val newPasswordHash = passwordService.hash(newPassword)
        val now = Instant.now()
        
        this.passwordHash = newPasswordHash
        this.updatedAt = now
        
        addDomainEvent(
            UserPasswordChangedEvent(
                userId = userId,
                newPasswordHash = newPasswordHash,
                changedAt = now
            )
        )
    }
    
    fun authenticate(password: String, passwordService: PasswordService): Boolean {
        return isActive && passwordService.verify(password, passwordHash)
    }
    
    // Getters
    fun getUserId(): UserId = userId
    fun getEmail(): Email = email
    fun getUsername(): Username = username
    fun getPasswordHash(): PasswordHash = passwordHash
    fun isActive(): Boolean = isActive
    fun getCreatedAt(): Instant = createdAt
    fun getUpdatedAt(): Instant = updatedAt
}
```

```kotlin
package com.lifee.user.domain.valueobjects

import com.lifee.user.domain.exceptions.InvalidEmailException
import java.util.regex.Pattern

data class Email(val value: String) {
    
    init {
        if (!isValid()) {
            throw InvalidEmailException(value)
        }
    }
    
    fun isValid(): Boolean {
        return EMAIL_PATTERN.matcher(value).matches()

    fun getDomain(): String {

        private val EMAIL_PATTERN = Pattern.compile(
            "^[A-Za-z0-9+_.-]+@[A-Za-z0-9.-]+\\.[A-Za-z]{2,}$"
        )
    }
```

```kotlin
// lifee-user-app/src/main/kotlin/com/lifee/user/application/commands/CreateUserCommand.kt
package com.lifee.user.application.commands

import com.lifee.common.cqrs.commands.Command
import com.lifee.user.domain.valueobjects.*

data class CreateUserCommand(
    val userId: UserId,
    val email: Email,
    val username: Username,
    val passwordHash: PasswordHash

// lifee-user-app/src/main/kotlin/com/lifee/user/application/commands/handlers/CreateUserCommandHandler.kt
package com.lifee.user.application.commands.handlers

import com.lifee.common.cqrs.commands.CommandHandler
import com.lifee.common.cqrs.events.EventBus
import com.lifee.user.application.commands.CreateUserCommand
import com.lifee.user.domain.aggregates.User
import com.lifee.user.domain.repositories.UserRepository
import com.lifee.user.domain.exceptions.DuplicateUserException
import org.springframework.stereotype.Component
import org.springframework.transaction.annotation.Transactional

@Component
class CreateUserCommandHandler(
    private val userRepository: UserRepository,
    private val eventBus: EventBus
) : CommandHandler<CreateUserCommand> {
    
    @Transactional
    override fun handle(command: CreateUserCommand) {
        // 检查用户是否已存在
        if (userRepository.existsByEmail(command.email)) {
            throw DuplicateUserException("Email already exists: ${command.email.value}")
        }
        
        if (userRepository.existsByUsername(command.username)) {
            throw DuplicateUserException("Username already exists: ${command.username.value}")
        }
        
        // 创建聚合根
        val user = User.create(
            userId = command.userId,
            email = command.email,
            username = command.username,
            passwordHash = command.passwordHash
        )
        
        // 保存聚合根
        userRepository.save(user)
        
        // 发布领域事件
        user.getDomainEvents().forEach { event ->
            eventBus.publish(event)
        }
        
        // 清除已发布的事件
        user.clearDomainEvents()
    }
}
```

```kotlin
import com.lifee.user.application.queries.GetUserQuery

import com.lifee.user.adapters.web.mappers.UserDtoMapper
import com.lifee.user.domain.exceptions.UserNotFoundException
import org.springframework.stereotype.Component

@Component
class GetUserQueryHandler(
    private val userJpaRepository: UserJpaRepository,
    private val userDtoMapper: UserDtoMapper
) : QueryHandler<GetUserQuery, UserResponse> {
```

### 5.2 自定义CQRS开发流程

#### 5.2.1 命令处理开发步骤

1. **定义命令**: 继承`Command`接口
```kotlin
data class CreateUserCommand(
    val userId: UserId,
    val email: Email
) : Command
```

2. **实现命令处理器**: 继承`CommandHandler`接口
```kotlin
@Component
class CreateUserCommandHandler : CommandHandler<CreateUserCommand> {
    override fun handle(command: CreateUserCommand) {
        // 处理逻辑
    }
}
```

3. **注册处理器**: 在配置类中注册
```kotlin
commandBus.registerHandler(CreateUserCommand::class, createUserCommandHandler)
```

#### 5.2.2 查询处理开发步骤

1. **定义查询**: 继承`Query`接口
2. **实现查询处理器**: 继承`QueryHandler`接口
3. **注册处理器**: 在配置类中注册

#### 5.2.3 事件处理开发步骤

1. **定义领域事件**: 继承`DomainEvent`基类

4. **注册处理器**: 在配置类中注册
